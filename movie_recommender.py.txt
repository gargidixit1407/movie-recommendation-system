movies = {
    "avengers": ["iron man", "thor", "captain america"],
    "batman": ["joker", "dark knight", "batman begins"],
    "spiderman": ["venom", "amazing spiderman", "homecoming"]
}

movie = input("Enter a movie name: ").lower()

if movie in movies:
    print("Recommended movies:")
    for m in movies[movie]:
        print("-", m)
else:
    print("Movie not found")
